from datasets import load_from_disk
import hashlib
import os
import pickle
import numpy as np

test = load_from_disk('local_dataset/test')
val = load_from_disk('local_dataset/val')
with open('embeddings/test_labels.pkl', 'rb') as handle:
    saved = pickle.load(handle)

def document_hash(record):
    return hashlib.sha256('\n'.join(record['text']).encode()).hexdigest()

test_hashes = {document_hash(test[i]) for i in range(3000)}
val_hashes = {document_hash(val[i]) for i in range(3000)}
score_matches = 0
label_matches = 0
text_score_length_matches = 0
text_label_length_matches = 0
for i in range(3000):
    score = np.asarray(test[i]['score'], dtype=float)
    label = np.asarray(test[i]['label'], dtype=float)
    archived = np.asarray(saved[i], dtype=float)
    score_matches += len(score) == len(archived) and np.allclose(
        score, archived, rtol=1e-6, atol=1e-7
    )
    label_matches += len(label) == len(archived) and np.array_equal(label, archived)
    text_score_length_matches += len(test[i]['text']) == len(score) == len(archived)
    text_label_length_matches += len(test[i]['text']) == len(label) == len(archived)

print('test_rows', len(test))
print('validation_rows', len(val))
print('saved_labels', len(saved))
print('val_test_exact_text_overlap_first_3000', len(test_hashes & val_hashes))
print('saved_labels_approx_match_score', score_matches)
print('saved_labels_exact_match_label', label_matches)
print('text_score_saved_equal_lengths', text_score_length_matches)
print('text_label_saved_equal_lengths', text_label_length_matches)
print('nonempty_saved_first_300', sum(any(float(v) >= .5 for v in saved[i]) for i in range(300)))
print('nonempty_saved_first_500', sum(any(float(v) >= .5 for v in saved[i]) for i in range(500)))
print('missing_test_emb_chunk_0', not os.path.exists('embeddings/test_emb_chunk_0.pt'))
print('missing_test_emb_chunk_1', not os.path.exists('embeddings/test_emb_chunk_1.pt'))
PY
